=== STEP 1: 1. Build cleaned skeleton multigraph with arc-length + tangent annotations ===
inputs:  skel (bool np.ndarray HxW), dist_map (float np.ndarray HxW from scipy.ndimage.distance_transform_edt).
outputs: G_skel: networkx.MultiGraph with annotated edges as described in data structures.
libs:    networkx.MultiGraph (already used). numpy for arc-length sum. Reuse existing penstroke.core.graph and penstroke.core.strokes.tangent_at.
algorithm:
from penstroke.core.graph import skeleton_to_graph, merge_nearby_junctions, collapse_parallel_edges
from penstroke.core.strokes import tangent_at
import numpy as np

def build_annotated_graph(skel, dist_map):
    G = skeleton_to_graph(skel)
    G = merge_nearby_junctions(G, max_dist=22)
    G = collapse_parallel_edges(G, dist_map)
    for u, v, k, d in G.edges(keys=True, data=True):
        path = d['path']
        # arc length
        arr = np.asarray(path, dtype=float)
        diffs = np.diff(arr, axis=0)
        d['length'] = float(np.hypot(diffs[:,0], diffs[:,1]).sum())
        # tangents (outward, 20-px baseline)
        d['tan_u'] = tangent_at(path, u, k=20)
        d['tan_v'] = tangent_at(path, v, k=20)
        d['retrace'] = False
    return G

=== STEP 2: 2. Split into connected components and detect orphaned closed loops ===
inputs:  G_skel, skel.
outputs: List[ComponentSpec] plus separately-collected closed_loop_strokes (list of Walks) from skeleton pixels not represented in G_skel.
libs:    networkx.connected_components, networkx.MultiGraph.subgraph. Reuse penstroke.core.strokes.trace_closed_loops for orphan loops.
algorithm:
import networkx as nx
from penstroke.core.strokes import trace_closed_loops

def split_components(G_skel, skel):
    components = []
    for nodes in nx.connected_components(G_skel):
        Gc = G_skel.subgraph(nodes).copy()
        if Gc.number_of_edges() == 0:
            continue  # isolated node, emit as dot later if needed
        odd = [n for n in Gc.nodes if Gc.degree(n) % 2 == 1]
        betti = Gc.number_of_edges() - Gc.number_of_nodes() + 1
        ys = [n[0] for n in Gc.nodes]; xs = [n[1] for n in Gc.nodes]
        total_len = sum(d['length'] for _,_,d in Gc.edges(data=True))
        spec = ComponentSpec(subgraph=Gc, odd_vertices=odd, betti=betti,
                             topology_class=classify(odd, betti, Gc),
                             bbox=(min(ys), min(xs), max(ys), max(xs)),
                             total_length=total_len)
        components.append(spec)
    # Orphaned closed loops (no special pixels) — 'o', 'O', counter of 'D'
    orphan_loops = trace_closed_loops(skel)
    return components, orphan_loops

def classify(odd, betti, Gc):
    if not odd and betti == 1: return 'CLOSED_LOOP'
    if not odd and betti >= 2: return 'MULTI_LOOP'
    if len(odd) == 2 and betti == 0: return 'OPEN_LINE'
    if len(odd) == 2 and betti >= 1: return 'LINE_WITH_LOOP'
    if len(odd) >= 4 and betti == 0: return 'TREE'
    return 'MIXED'

import numpy as np

def _angle(t_in, t_out_outward):
    # t_in is tangent ARRIVING at node (pointing toward node).
    # t_out_outward is tangent LEAVING node (pointing away from node, as stored).
    # Straight-through means t_in and t_out_outward point in the SAME direction:
    #   dot(t_in, t_out_outward) close to +1 -> straight.
    #   dot close to -1 -> U-turn.
    d = float(np.clip(np.dot(t_in, t_out_outward), -1.0, 1.0))
    return np.arccos(d)  # 0 = straight, pi = u-turn

=== STEP 6: 6. Closed-component handling (CLOSED_LOOP and MULTI_LOOP) ===
inputs:  ComponentSpec where odd_vertices == [], or orphan_loops from step 2.
outputs: list[Walk] with closed walks rotated to start at topmost vertex and oriented clockwise.
libs:    numpy.argmin, numpy.sum for shoelace, reuse trace_closed_loops for orphan-loop pixel walks.
algorithm:
import numpy as np

def handle_closed_component(spec):
    Gc = spec.subgraph
    if Gc.number_of_nodes() == 0:
        return []
    # Pick deterministic start: topmost (min y) then leftmost (min x).
    start = min(Gc.nodes, key=lambda n: (n[0], n[1]))
    G_aug = Gc.copy()
    trail = hierholzer_continuity(G_aug, start)
    walks = trail_to_pixels_and_split(trail, G_aug)
    # If no sharp turn fired and we have one closed walk, ensure clockwise orientation.
    walks = [orient_clockwise_if_closed(w) for w in walks]
    return walks

def orient_clockwise_if_closed(walk):
    if walk[0] != walk[-1]:
        return walk
    arr = np.asarray(walk, dtype=float)
    # Shoelace in image coords (y goes DOWN): positive = clockwise visually.
    x = arr[:,1]; y = arr[:,0]
    area2 = float(np.sum(x[:-1]*y[1:] - x[1:]*y[:-1]))
    if area2 < 0:           # counter-clockwise in image coords
        walk = walk[::-1]
    # Rotate so topmost pixel of the closed walk is first.
    top_idx = int(np.argmin([p[0] for p in walk[:-1]]))
    walk = walk[top_idx:-1] + walk[:top_idx] + [walk[top_idx]]
    return walk

def handle_orphan_loop(pixel_list):
    # Already comes from trace_closed_loops which rotates+orients; just pass through.
    return pixel_list

def trace_glyph_eulerian(skel, dist_map):
    G = build_annotated_graph(skel, dist_map)
    specs, orphan_loops = split_components(G, skel)
    components_walks = []
    for spec in specs:
        if spec.topology_class in ('CLOSED_LOOP', 'MULTI_LOOP'):
            components_walks.append(handle_closed_component(spec))
        else:
            components_walks.append(handle_open_component(spec))
    ordered = order_all_walks(components_walks, orphan_loops, specs)
    # Drop walks shorter than the minimum stroke length (skeleton noise).
    return [w for w in ordered if polyline_length(w) >= MIN_WALK_LEN_PX]
